NOTE: You have not set ssh.existingSecret. The pod cannot open the SSH tunnel to
the dev box, so the Traefik primary will never come up. Create the Secret and
enable the ssh-key volume:

  ssh-keygen -t ed25519 -N '' -f ~/.ssh/collie-waker
  ssh-keyscan -H {{ required "ssh.host is required" .Values.ssh.host }} > known_hosts

  kubectl create secret generic collie-waker-ssh-key \
    --namespace {{ .Release.Namespace }} \
    --from-file=ssh-privatekey=~/.ssh/collie-waker \
    --from-file=known_hosts=known_hosts

Append the public key to ~{{ .Values.ssh.user }}/.ssh/authorized_keys on the dev
box, then enable the volume and upgrade:

  helm upgrade {{ .Release.Name }} ./charts/collie-waker \
    --reuse-values \
    --set ssh.existingSecret=collie-waker-ssh-key \
    --set persistence.ssh-key.enabled=true

{{- end }}
